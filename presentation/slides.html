export function loadTafsirData() {
  if (!dataPromise) {
    dataPromise = import('./tafsir')
      .then(m => m.TAFSIR_DATA)
      .catch(err => {
        dataPromise = null;
        throw err;

الحد الأمني

من يعرف UUID يستطيع انتحال الجهاز. هذا مقبول لبيانات قراءة غير حساسة، وليس بديلاً عن auth أو RLS.

src/app/api/user-data/route.ts
import { NextResponse, type NextRequest } from 'next/server';
import { getSupabase } from '@/src/lib/supabase';

function getDeviceId(req: NextRequest) {
  return req.headers.get('x-device-id');
}

export async function PUT(req: NextRequest) {
  const deviceId = getDeviceId(req);
  if (!deviceId) return NextResponse.json(
    { error: 'Missing x-device-id header' }, { status: 400 }
  );

  let body: { bookmarks?: unknown[]; history?: unknown[]; completed?: number[]; theme?: string };
  try { body = await req.json(); }
  catch { return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 }); }

  const { bookmarks, history, completed, theme } = body;
  const { data, error } = await getSupabase().from('user_data')
    .upsert({ device_id: deviceId, bookmarks: bookmarks ?? [], history: history ?? [], completed: completed ?? [], theme: theme ?? 'dark' }, { onConflict: 'device_id' }).select().single();
  if (error) return NextResponse.json({ error: error.message }, { status: 500 });
  return NextResponse.json(data);
}
